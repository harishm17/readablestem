\section*{Homework 3: linear algebra}

\begin{problem}{1}
For $\vec{x}, \vec{y} \in \R^n$ show that $|\inner{\vec{x}}{\vec{y}}| \le \norm{\vec{x}}\,\norm{\vec{y}}$. \points{5}
\end{problem}

\begin{problem}{2}
Show that $\tr(AB) = \tr(BA)$ and find $\argmin_{\vec{w}} \norm{X\vec{w} - \vec{y}}^2$. \points{10}
\end{problem}

\begin{problem}{3}
Let $\eps > 0$. If $f(t) = e^{-t^2}$, compute $\dd[f]{t}$ and show that $|f(t)| < \eps$ for large $t$. This is \emph{important}. \points{5}
\end{problem}
